#
# Raspberry Pi firmware boot configuration for pi-cannonball.
#
# One file for all three boards. The firmware reads the model section that
# matches the board it is running on and boots the image named there, so a
# card or a netboot directory holding all three images serves any of them.
#
# Derived from Circle's boot/config64.txt. The Compute Modules are pinned to
# their board's image explicitly: stock config64.txt leaves [cm4] with only
# otg_mode=1 and no kernel line, so a CM4 — the same silicon as a Pi 4 —
# falls through to the default kernel8.img, which is the Pi 3 build, and runs
# the wrong architecture.
#

arm_64bit=1
kernel_address=0x80000
initial_turbo=0

# Overscan is the firmware inset that keeps a picture clear of a CRT's
# hidden edges: it shrinks the output and surrounds it with black. Every
# display this reaches is digital and shows its whole signal, so the inset
# only ever costs picture — measured on a Pi 4 as a border on all four
# sides, present in the boot console as well as the game.
disable_overscan=1

# --- Pi 3 world (BCM2837): kernel8.img ---

# Zero 2 W
[pi02]
kernel=kernel8.img

# Pi 2 v1.2 (BCM2837, the only 64-bit-capable Pi 2)
[pi2]
kernel=kernel8.img

[pi3]
kernel=kernel8.img

[pi3+]
kernel=kernel8.img

# Compute Module 3
[cm3]
kernel=kernel8.img

# --- Pi 4 world (BCM2711): kernel8-rpi4.img ---

[pi4]
armstub=armstub8-rpi4.bin
kernel=kernel8-rpi4.img
max_framebuffers=2

# Compute Module 4 — same silicon as the Pi 4
[cm4]
otg_mode=1
armstub=armstub8-rpi4.bin
kernel=kernel8-rpi4.img
max_framebuffers=2

# --- Pi 5 world (BCM2712): kernel_2712.img ---

[pi5]
kernel=kernel_2712.img
# Nothing here sets the Pi 5's resolution, because nothing can. Forcing 720p
# with every lever the firmware offers — hdmi_force_hotplug, hdmi_group=1,
# hdmi_mode=4, framebuffer_width and framebuffer_height, and Circle's own
# width= and height= alongside them — left the board still emitting 1080p.
# It also grants a kernel only the rows the mode it already chose has,
# whatever virtual size the mailbox asks for, which is why the shim keeps a
# single-surface present path for this board.
#
# Nor is the mode simply the panel's own. On the bench a Pi 5 drove 1080p
# into a capture dongle and into a television, and produced no picture at all
# on a 1024x768 panel and an 800x480 one that a Pi 3 and a Pi 4 both lit from
# this same card. A card carrying that forced-720p configuration did get a
# picture onto those two — a stretched 1080p — so the board can be made to
# drive them, badly; which of the levers did it was not isolated.
#
# Depth is set because everything circle-libsdl2 renders is 32-bit pixels and
# the Pi 5 build is compiled to match. The board has only ever been run with
# this line present, so what the firmware would settle on without it is
# untested. Pi 3 and Pi 4 take their surface from the mailbox and are
# unaffected by both lines.
framebuffer_depth=32
framebuffer_ignore_alpha=1

# Compute Module 5
[cm5]
kernel=kernel_2712.img
framebuffer_depth=32
framebuffer_ignore_alpha=1
